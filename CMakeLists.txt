cmake_minimum_required(VERSION 3.24)
project(wray VERSION 1.0.5 LANGUAGES CXX)

if(PROJECT_IS_TOP_LEVEL AND NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

if(PROJECT_IS_TOP_LEVEL AND NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 23)
endif()

option(WRAY_BUILD_TESTS "Build tests" ${PROJECT_IS_TOP_LEVEL})
option(WRAY_INSTALL "Generate install rules and a CMake package" ${PROJECT_IS_TOP_LEVEL})
option(WRAY_BUILD_BENCH "Build benchmarks" ${PROJECT_IS_TOP_LEVEL})
option(WRAY_BUILD_TOOLS "Build tools" ${PROJECT_IS_TOP_LEVEL})
option(WRAY_STRICT_FP "Forbid the compiler from fusing or reassociating floating-point operations in consumers" ON)
set(WRAY_SIMD "auto" CACHE STRING "auto, scalar, sse2, avx2 or avx512")
set_property(CACHE WRAY_SIMD PROPERTY STRINGS auto scalar sse2 avx2 avx512)
if(NOT WRAY_SIMD MATCHES "^(auto|scalar|sse2|avx2|avx512)$")
    message(FATAL_ERROR "WRAY_SIMD must be auto, scalar, sse2, avx2 or avx512, got ${WRAY_SIMD}")
endif()

add_library(wray INTERFACE)
add_library(wray::wray ALIAS wray)
target_include_directories(wray INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(wray INTERFACE cxx_std_20)
if(WRAY_STRICT_FP)
    if(MSVC)
        target_compile_options(wray INTERFACE /fp:precise)
    else()
        target_compile_options(wray INTERFACE -ffp-contract=off)
    endif()
endif()
if(WRAY_SIMD STREQUAL "scalar")
    target_compile_definitions(wray INTERFACE WRAY_SIMD_SCALAR=1)
elseif(NOT WRAY_SIMD STREQUAL "auto")
    set(wray_target_arch "${CMAKE_SYSTEM_PROCESSOR}")
    if(APPLE AND CMAKE_OSX_ARCHITECTURES)
        set(wray_target_arch "${CMAKE_OSX_ARCHITECTURES}")
    endif()
    if(NOT wray_target_arch MATCHES "^(x86_64|AMD64|amd64|x64|i[3-6]86|x86)$")
        message(FATAL_ERROR "WRAY_SIMD=${WRAY_SIMD} needs an x86 target, got ${wray_target_arch}")
    endif()
    if(MSVC)
        if(WRAY_SIMD STREQUAL "avx2")
            target_compile_options(wray INTERFACE /arch:AVX2)
        elseif(WRAY_SIMD STREQUAL "avx512")
            target_compile_options(wray INTERFACE /arch:AVX512)
        endif()
    elseif(WRAY_SIMD STREQUAL "sse2")
        target_compile_options(wray INTERFACE -msse2 -mno-avx)
    elseif(WRAY_SIMD STREQUAL "avx2")
        target_compile_options(wray INTERFACE -mavx2 -mfma)
    else()
        target_compile_options(wray INTERFACE -mavx512f -mavx2 -mfma)
    endif()
endif()

add_library(wray_warnings INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(wray_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Werror)
endif()

if(WRAY_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

if(WRAY_BUILD_TOOLS)
    add_subdirectory(tools)
endif()

if(WRAY_BUILD_BENCH)
    add_subdirectory(bench)
endif()

if(WRAY_INSTALL)
    include(GNUInstallDirs)
    include(CMakePackageConfigHelpers)
    install(TARGETS wray EXPORT wrayTargets)
    install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
    install(EXPORT wrayTargets NAMESPACE wray:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/wray)
    configure_package_config_file(cmake/wrayConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/wrayConfig.cmake"
                                  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/wray)
    write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/wrayConfigVersion.cmake" COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/wrayConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/wrayConfigVersion.cmake"
            DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/wray)
endif()
