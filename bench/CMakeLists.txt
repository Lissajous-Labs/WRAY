add_executable(wray_bench bench.cpp libm.cpp)
target_link_libraries(wray_bench PRIVATE wray::wray)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(wray_bench_scalar_options -fno-vectorize -fno-slp-vectorize)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set(wray_bench_scalar_options -fno-tree-vectorize)
elseif(MSVC)
    set(wray_bench_scalar_options /Qvec-)
else()
    message(FATAL_ERROR "The benchmark needs an explicit scalar-vectorization control for this compiler")
endif()
set_source_files_properties(libm.cpp PROPERTIES COMPILE_OPTIONS "${wray_bench_scalar_options}")
string(JOIN " " wray_bench_scalar_flags ${wray_bench_scalar_options})
if(APPLE)
    target_sources(wray_bench PRIVATE vforce.cpp)
    target_link_libraries(wray_bench PRIVATE "-framework Accelerate")
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" wray_bench_config)
target_compile_definitions(wray_bench PRIVATE
    WRAY_BENCH_FLAGS="${CMAKE_CXX_COMPILER_ID} ${CMAKE_BUILD_TYPE} ${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${wray_bench_config}} CXX_STANDARD=${CMAKE_CXX_STANDARD} WRAY_SIMD=${WRAY_SIMD} $<JOIN:$<TARGET_PROPERTY:wray,INTERFACE_COMPILE_OPTIONS>, >"
    WRAY_BENCH_LIBM_FLAGS="${wray_bench_scalar_flags}")
